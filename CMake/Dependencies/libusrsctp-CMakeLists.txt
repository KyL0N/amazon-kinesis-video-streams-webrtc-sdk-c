cmake_minimum_required(VERSION 3.6.3)

project(libusrsctp-download NONE)

include(ExternalProject)

# The frozen 2021 baseline still declares compatibility with CMake 2.8. CMake
# 4 removes that compatibility mode before the dependency can configure. Keep
# the source revision unchanged and opt the child build into the oldest policy
# level CMake 4 still accepts. Older CMake versions used by the ARM targets do
# not need or understand this cache variable.
set(USRSCTP_CMAKE_POLICY_ARGS "")
if(CMAKE_VERSION VERSION_GREATER_EQUAL "4.0")
    list(APPEND USRSCTP_CMAKE_POLICY_ARGS -DCMAKE_POLICY_VERSION_MINIMUM=3.5)
endif()

if(KVS_USRSCTP_SOURCE_DIR)
    set(USRSCTP_DOWNLOAD_ARGS
        SOURCE_DIR       "${KVS_USRSCTP_SOURCE_DIR}"
        DOWNLOAD_COMMAND ""
        UPDATE_COMMAND   "")
else()
    if(NOT KVS_USRSCTP_GIT_REPOSITORY)
        set(KVS_USRSCTP_GIT_REPOSITORY https://github.com/sctplab/usrsctp.git)
    endif()
    if(NOT KVS_USRSCTP_GIT_TAG)
        set(KVS_USRSCTP_GIT_TAG 1ade45cbadfd19298d2c47dc538962d4425ad2dd)
    endif()
    set(USRSCTP_DOWNLOAD_ARGS
        GIT_REPOSITORY ${KVS_USRSCTP_GIT_REPOSITORY}
        GIT_TAG        ${KVS_USRSCTP_GIT_TAG}
        GIT_PROGRESS   TRUE)
endif()

ExternalProject_Add(project_libusrsctp
    ${USRSCTP_DOWNLOAD_ARGS}
    PREFIX            ${CMAKE_CURRENT_BINARY_DIR}/build
    CMAKE_ARGS        -DCMAKE_INSTALL_PREFIX=${OPEN_SRC_INSTALL_PREFIX}
                      -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
                      "-DCMAKE_C_FLAGS=${CMAKE_C_FLAGS} -fPIC"
                      -Dsctp_werror=0
                      -Dsctp_build_programs=0
                      -Dsctp_af_conn_output_scratch=${KVS_USRSCTP_AF_CONN_OUTPUT_SCRATCH}
                      -Dsctp_borrow_sack_input=${KVS_USRSCTP_BORROW_SACK_INPUT}
                      -Dsctp_fast_path_stats=${KVS_USRSCTP_FAST_PATH_STATS}
                      ${USRSCTP_CMAKE_POLICY_ARGS}
    BUILD_ALWAYS      TRUE
    TEST_COMMAND      ""
)
